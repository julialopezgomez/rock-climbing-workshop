cmake_minimum_required(VERSION 3.16)
project(MicrobanM3T LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(OpenGL_GL_PREFERENCE LEGACY)
find_package(OpenMP REQUIRED)
find_package(Eigen3 3.3 REQUIRED)
find_package(GLEW REQUIRED)
find_package(OpenGL REQUIRED)
find_package(glfw3 REQUIRED)
find_package(OpenCV 4.3 REQUIRED COMPONENTS core imgproc highgui imgcodecs calib3d features2d xfeatures2d)
if(NOT M3T_ROOT OR NOT M3T_LIBRARY)
  message(FATAL_ERROR "Set M3T_ROOT and M3T_LIBRARY to the isolated upstream build")
endif()
find_library(EGL_LIBRARY EGL REQUIRED)
add_executable(microban_m3t main.cpp headless.cpp)
target_include_directories(microban_m3t PRIVATE "${M3T_ROOT}/include" "${M3T_ROOT}/third_party" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(microban_m3t PRIVATE "${M3T_LIBRARY}" Eigen3::Eigen OpenGL::GL GLEW::GLEW glfw ${EGL_LIBRARY} ${CMAKE_DL_LIBS} ${OpenCV_LIBRARIES} OpenMP::OpenMP_CXX pthread stdc++fs)
